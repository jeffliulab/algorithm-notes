这两个模板一定要区分好

```python
class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        if not nums:
            return [-1,-1]
    
        # 两种模板

        # 第一种： 
        left = 0
        right = len(nums)-1  # 这里是 -1 
        while left <= right:
            mid = (left+right)//2
            if mid == target:
                # find the target
            elif mid < target:
                left = mid + 1
            else:
                # 缩上界，因为右边是闭区间
                right = mid - 1
        # 循环结束后
        # 这个时候left和right正好差1，且left = right + 1
    

        # 第二种：
        left = 0
        right = len(nums)
        while left < right:
            mid = (left+right)//2
            if mid == target:
                # find the target
            elif mid < target:
                left = mid + 1
            else:
                # 缩上界，因为右边是开区间
                right = mid
        # 循环结束后
        # 这个时候left和right正好相等









```
